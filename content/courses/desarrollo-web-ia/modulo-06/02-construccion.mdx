## El plan del día

Con el brief, schema y CLAUDE.md listos, la construcción sigue este orden:

```
Bloque 1 (2 horas): Infraestructura
  → DB + migraciones + cliente Drizzle
  → Supabase Auth + middleware
  → Páginas de login/registro
  → Layout autenticado

Bloque 2 (3 horas): Core features
  → CRUD de la entidad principal
  → Server Actions o API Routes
  → UI con shadcn/ui

Bloque 3 (2 horas): Polish + Deploy
  → Manejo de errores
  → Loading states
  → Deploy a Vercel
```

No intentes construir todo en un bloque. Cada bloque tiene un entregable funcional.

### El middleware que necesitas

```typescript
// middleware.ts
import { createServerClient } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

const PUBLIC_ROUTES = ['/', '/auth/login', '/auth/registro', '/auth/recuperar']
const AUTH_ROUTES = ['/auth/login', '/auth/registro', '/auth/recuperar']

export async function middleware(request: NextRequest) {
  let supabaseResponse = NextResponse.next({ request })

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() { return request.cookies.getAll() },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value))
          supabaseResponse = NextResponse.next({ request })
          cookiesToSet.forEach(({ name, value, options }) =>
            supabaseResponse.cookies.set(name, value, options)
          )
        },
      },
    }
  )

  const { data: { user } } = await supabase.auth.getUser()
  const path = request.nextUrl.pathname

  // Si no autenticado y quiere ruta protegida → login
  if (!user && !PUBLIC_ROUTES.includes(path)) {
    const url = request.nextUrl.clone()
    url.pathname = '/auth/login'
    url.searchParams.set('redirectTo', path)
    return NextResponse.redirect(url)
  }

  // Si autenticado y quiere ruta de auth → app
  if (user && AUTH_ROUTES.includes(path)) {
    return NextResponse.redirect(new URL('/app', request.url))
  }

  return supabaseResponse
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico|api/webhooks).*)'],
}
```

## Bloque 2: CRUD con Server Actions

Las Server Actions son la forma más directa de hacer CRUD en Next.js 15 — sin necesidad de crear API routes separadas para operaciones simples.

### Estructura de un Server Action

```typescript
// src/app/(app)/projects/actions.ts
'use server'

import { revalidatePath } from 'next/cache'
import { db } from '@/db'
import { projects } from '@/db/schema'
import { createServerClient } from '@/lib/supabase/server'
import { eq, and } from 'drizzle-orm'
import { z } from 'zod'

const createProjectSchema = z.object({
  name: z.string().min(1).max(100),
  description: z.string().optional(),
  color: z.string().regex(/^#[0-9a-fA-F]{6}$/).optional(),
})

type ActionResult<T = void> =
  | { success: true; data: T }
  | { success: false; error: string }

export async function createProject(
  formData: FormData
): Promise<ActionResult<typeof projects.$inferSelect>> {
  // 1. Obtener usuario actual
  const supabase = createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { success: false, error: 'No autenticado' }

  // 2. Validar input
  const parsed = createProjectSchema.safeParse({
    name: formData.get('name'),
    description: formData.get('description'),
    color: formData.get('color'),
  })
  if (!parsed.success) return { success: false, error: parsed.error.issues[0].message }

  // 3. Crear en DB
  const [project] = await db.insert(projects).values({
    userId: user.id,
    ...parsed.data,
  }).returning()

  // 4. Revalidar cache
  revalidatePath('/app')
  revalidatePath('/app/projects')

  return { success: true, data: project }
}

export async function deleteProject(projectId: string): Promise<ActionResult> {
  const supabase = createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { success: false, error: 'No autenticado' }

  // Verificar ownership antes de eliminar
  await db.delete(projects).where(
    and(
      eq(projects.id, projectId),
      eq(projects.userId, user.id) // nunca fiar del cliente
    )
  )

  revalidatePath('/app/projects')
  return { success: true, data: undefined }
}
```

## Bloque 3: UI conectada a datos reales

### Patrón: Server Component + Client Form

```typescript
// src/app/(app)/projects/page.tsx — Server Component (lee datos)
import { db } from '@/db'
import { projects, tasks } from '@/db/schema'
import { eq, sql, desc } from 'drizzle-orm'
import { createServerClient } from '@/lib/supabase/server'
import { ProjectList } from './_components/ProjectList'
import { CreateProjectButton } from './_components/CreateProjectButton'

export default async function ProjectsPage() {
  const supabase = createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return null // el middleware ya redirige

  const projectsWithCount = await db
    .select({
      project: projects,
      taskCount: sql<number>`count(${tasks.id})::int`,
      completedCount: sql<number>`count(case when ${tasks.isCompleted} then 1 end)::int`,
    })
    .from(projects)
    .leftJoin(tasks, eq(tasks.projectId, projects.id))
    .where(eq(projects.userId, user.id))
    .groupBy(projects.id)
    .orderBy(desc(projects.createdAt))

  return (
    <main className="p-6 max-w-5xl mx-auto">
      <div className="flex items-center justify-between mb-8">
        <h1 className="text-3xl font-bold">Mis Proyectos</h1>
        <CreateProjectButton />
      </div>
      <ProjectList projects={projectsWithCount} />
    </main>
  )
}
```

```typescript
// src/app/(app)/projects/_components/CreateProjectButton.tsx — Client Component
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { Button } from '@/components/ui/button'
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import { createProject } from '../actions'
import { Plus, Loader2 } from 'lucide-react'

export function CreateProjectButton() {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)
  const router = useRouter()

  async function handleSubmit(formData: FormData) {
    setLoading(true)
    const result = await createProject(formData)
    setLoading(false)

    if (result.success) {
      setOpen(false)
      router.push(`/app/projects/${result.data.id}`)
    } else {
      // mostrar error
    }
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>
        <Button><Plus className="w-4 h-4 mr-2" />Nuevo proyecto</Button>
      </DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Crear proyecto</DialogTitle>
        </DialogHeader>
        <form action={handleSubmit} className="space-y-4">
          <div>
            <Label htmlFor="name">Nombre</Label>
            <Input id="name" name="name" required placeholder="Mi proyecto" />
          </div>
          <Button type="submit" disabled={loading} className="w-full">
            {loading ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null}
            Crear
          </Button>
        </form>
      </DialogContent>
    </Dialog>
  )
}
```

## El prompt de construcción de un feature completo

```
Necesito implementar el feature de [nombre del feature].

Schema involucrado:
[tablas relevantes del schema]

Funcionalidad esperada:
1. [Comportamiento 1]
2. [Comportamiento 2]
3. [Comportamiento 3]

Estructura a crear:
- src/app/(app)/[ruta]/page.tsx — Server Component, lee datos
- src/app/(app)/[ruta]/actions.ts — Server Actions (create, update, delete)
- src/app/(app)/[ruta]/_components/ — Client Components para interactividad

Reglas:
- Server Components para lectura de datos (no useEffect + fetch)
- Server Actions para mutaciones (no API routes para CRUD simple)
- Manejo de errores: mostrar toast en error, redirigir en éxito
- Loading states mientras procesa la mutación
- Validación con Zod en el server action
```

Este prompt genera el feature completo (server + client + actions) en una sola respuesta.
